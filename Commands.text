Deploy any Yaml : Kubectl apply -f namespace.yaml

get pods - kubectl get pods

get namespaces - kubectl get ns

exec into a pod : kubectl exec -it my-pod -n sunil-namespace -- bash

describe a pod : kubectl describe pod my-pod -n default

get all deployment : kubectl get deployment -n default

scale a deployment : kubectl scale deployment nginx-deployment -n sunil-namespace --replicas=4

list all configmaps: kubectl get configmap -n dafault

taint a node :  kubectl taint node sunil-cluster-worker    prod=true:NoSchedule
